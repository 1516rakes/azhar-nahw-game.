    class SoundEngine {
      constructor() {
        this.enabled = true;
        this.ctx = null;
      }

      init() {
        try {
          if (!this.ctx) {
            const AudioCtx = window.AudioContext || window.webkitAudioContext;
            if (AudioCtx) this.ctx = new AudioCtx();
          }
          if (this.ctx && this.ctx.state === 'suspended') {
            this.ctx.resume();
          }
        } catch (e) {}
      }

      playTone(freq, type, duration, vol = 0.15, slideFreq = null) {
        if (!this.enabled) return;
        this.init();
        if (!this.ctx) return;

        try {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = type;
          osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
          if (slideFreq) {
            osc.frequency.exponentialRampToValueAtTime(slideFreq, this.ctx.currentTime + duration);
          }

          gain.gain.setValueAtTime(vol, this.ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);

          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start();
          osc.stop(this.ctx.currentTime + duration);
        } catch (e) {}
      }

      cast() { this.playTone(520, 'sine', 0.25, 0.14, 180); }
      bite() { this.playTone(320, 'triangle', 0.15, 0.16, 620); }
      correct() {
        this.playTone(523.25, 'sine', 0.14, 0.18);
        setTimeout(() => this.playTone(659.25, 'sine', 0.14, 0.18), 100);
        setTimeout(() => this.playTone(783.99, 'sine', 0.28, 0.2), 200);
      }
      wrong() { this.playTone(260, 'sawtooth', 0.22, 0.12, 140); }
      buy() {
        this.playTone(700, 'triangle', 0.12, 0.18);
        setTimeout(() => this.playTone(980, 'triangle', 0.22, 0.2), 90);
      }
    }

    /**
     * =========================================================================
     * 4. تطبيع ومطابقة الإجابات العربية المكتوبة يدوياً
     * =========================================================================

    function normalizeArabicText(str) {
      if (!str) return "";
      return str
        .toString()
        .trim()
        // إزالة التشكيل والمد
        .replace(/[\u064B-\u065F\u0670\u0640]/g, "")
        // توحيد الهمزات
        .replace(/[أإآٱ]/g, "ا")
        // توحيد التاء المربوطة والهاء
        .replace(/ة/g, "ه")
        // توحيد الياء والألف المقصورة
        .replace(/ى/g, "ي")
        // إزالة علامات الترقيم والرموز الزائدة
        .replace(/[.,،؛؟!:"'()\[\]{}\-_/\\]/g, " ")
        // توحيد المسافات المتعددة
        .replace(/\s+/g, " ")
        .trim()
        .toLowerCase();
    }

    function stripAlPrefix(normStr) {
      return normStr.startsWith("ال") && normStr.length > 3 ? normStr.slice(2) : normStr;
    }

    function checkWrittenAnswerMatch(userInput, acceptedAnswers) {
      const normUser = normalizeArabicText(userInput);
      if (!normUser) return false;
      const userNoAl = stripAlPrefix(normUser);

      return acceptedAnswers.some(ans => {
        const normAns = normalizeArabicText(ans);
        const ansNoAl = stripAlPrefix(normAns);
        return (
          normUser === normAns ||
          userNoAl === ansNoAl ||
          (normAns.length >= 3 && normUser.includes(normAns))
        );
      });
    }

    /**
     * =========================================================================
     * 8. فيزياء تأرجح الصنارة البندولية الملتصقة بطرف العصا
     * =========================================================================
     */
    function getRodTipPlayAreaCoords() {
      const boatSvg = document.getElementById("boatSvg");
      const playArea = document.getElementById("playArea");
      if (!boatSvg || !playArea) return { x: window.innerWidth / 2, y: 120 };

      const svgRect = boatSvg.getBoundingClientRect();
      const areaRect = playArea.getBoundingClientRect();

      if (svgRect.width === 0 || areaRect.width === 0) {
        return { x: window.innerWidth / 2, y: 120 };
      }

      const tipX = (svgRect.left - areaRect.left) + (178 / 240) * svgRect.width;
      const tipY = (svgRect.top - areaRect.top) + (10 / 130) * svgRect.height;
      return { x: tipX, y: tipY };
    }

    function updateHookVisuals() {
      const tip = getRodTipPlayAreaCoords();
      const hookX = tip.x + Math.sin(hookPhysics.angle) * hookPhysics.currentLength;
      const hookY = tip.y + Math.cos(hookPhysics.angle) * hookPhysics.currentLength;

      const line = document.getElementById("fishingLine");
      const hook = document.getElementById("hookGroup");

      if (line && hook) {
        line.setAttribute("x1", tip.x);
        line.setAttribute("y1", tip.y);
        line.setAttribute("x2", hookX);
        line.setAttribute("y2", hookY);

        const deg = (-hookPhysics.angle * 180) / Math.PI;
        hook.setAttribute("transform", `translate(${hookX}, ${hookY}) rotate(${deg})`);
      }

      return { tipX: tip.x, tipY: tip.y, hookX, hookY };
    }

    function launchHook() {
      if (hookPhysics.state !== 'swinging' || gameState.isCasting) return;
      if (activeFishes.length === 0) return;
      hookPhysics.state = 'shooting';
      hookPhysics.caughtFishes = [];
      gameState.isCasting = true;
      sound.cast();
    }

    function animateFishes() {
      const playArea = document.getElementById("playArea");
      const oceanEl = document.getElementById("oceanArea");

      if (playArea && oceanEl) {
        const areaRect = playArea.getBoundingClientRect();
        const oceanRect = oceanEl.getBoundingClientRect();
        const maxWidth = oceanEl.clientWidth || window.innerWidth;
        const oceanOffsetY = oceanRect.top - areaRect.top;

        // 1. تحريك الأسماك أفقياً
        activeFishes.forEach(fish => {
          if (fish.caught || hookPhysics.caughtFishes.includes(fish)) return;

          fish.x += fish.vx;
          if (fish.x <= 8) {
            fish.x = 8;
            fish.vx = Math.abs(fish.vx);
            fish.svgWrapper.style.transform = "scaleX(1)";
          } else if (fish.x + fish.width >= maxWidth - 8) {
            fish.x = maxWidth - fish.width - 8;
            fish.vx = -Math.abs(fish.vx);
            fish.svgWrapper.style.transform = "scaleX(-1)";
          }

          fish.el.style.left = fish.x + "px";
        });

        // 2. تحديث حركة الصنارة البندولية
        const rodSpeedBoost = 1 + (gameState.upgrades.rodLevel - 1) * 0.28;

        if (hookPhysics.state === 'swinging') {
          hookPhysics.angle += 0.025 * hookPhysics.swingDir;
          if (hookPhysics.angle >= 1.15) {
            hookPhysics.angle = 1.15;
            hookPhysics.swingDir = -1;
          } else if (hookPhysics.angle <= -1.15) {
            hookPhysics.angle = -1.15;
            hookPhysics.swingDir = 1;
          }
          hookPhysics.currentLength = hookPhysics.baseLength;
        } else if (hookPhysics.state === 'shooting') {
          hookPhysics.currentLength += 9.5 * rodSpeedBoost;
          const coords = updateHookVisuals();

          const maxDepthY = oceanOffsetY + (getMaxDepthPercent() / 100) * oceanRect.height;
          const outOfBounds =
            coords.hookX <= 12 ||
            coords.hookX >= areaRect.width - 12 ||
            coords.hookY >= areaRect.height - 15 ||
            coords.hookY >= maxDepthY;

          const hitRadius = gameState.upgrades.doubleCatch === 1 ? 46 : 32;
          activeFishes.forEach(fish => {
            if (fish.caught || hookPhysics.caughtFishes.includes(fish)) return;
            const fishCenterX = fish.x + fish.width / 2;
            const fishCenterY = oceanOffsetY + (fish.yPct / 100) * oceanRect.height + fish.height / 2 + 12;

            const dist = Math.hypot(coords.hookX - fishCenterX, coords.hookY - fishCenterY);
            if (dist <= hitRadius) {
              const maxAllowedCatch = gameState.upgrades.doubleCatch === 1 ? 2 : 1;
              if (hookPhysics.caughtFishes.length < maxAllowedCatch) {
                hookPhysics.caughtFishes.push(fish);
                fish.el.classList.add("scale-125");
                sound.bite();
              }
            }
          });

          const maxAllowedCatch = gameState.upgrades.doubleCatch === 1 ? 2 : 1;
          if (hookPhysics.caughtFishes.length >= maxAllowedCatch || outOfBounds) {
            if (hookPhysics.caughtFishes.length === 0 && coords.hookY >= maxDepthY && getMaxDepthPercent() < 98) {
              showToast("📏 وصلت الصنارة لأقصى مسافة! طوّر طول الخيط من المتجر للوصول للأعماق.");
            }
            hookPhysics.state = 'retracting';
          }
        } else if (hookPhysics.state === 'retracting') {
          const retractSpeed = hookPhysics.caughtFishes.length > 0 ? 7.5 * rodSpeedBoost : 12 * rodSpeedBoost;
          hookPhysics.currentLength -= retractSpeed;
          const coords = updateHookVisuals();

          hookPhysics.caughtFishes.forEach((fish, idx) => {
            const offsetX = idx === 0 ? 0 : 22;
            fish.x = coords.hookX - fish.width / 2 + offsetX;
            fish.el.style.left = fish.x + "px";
            fish.el.style.top = Math.max(0, coords.hookY - oceanOffsetY - fish.height / 2) + "px";
          });

          if (hookPhysics.currentLength <= hookPhysics.baseLength) {
            hookPhysics.currentLength = hookPhysics.baseLength;
            hookPhysics.state = 'swinging';

            if (hookPhysics.caughtFishes.length > 0) {
              gameState.activeFishGroup = [...hookPhysics.caughtFishes];
              openQuestionModal(gameState.activeFishGroup);
            } else {
              gameState.isCasting = false;
            }
          }
        }

        updateHookVisuals();
      }

      animationFrameId = requestAnimationFrame(animateFishes);
    }

    /**
     * =========================================================================
     * 9. نافذة الأسئلة النحوية بأنواعها الثلاثة (اختيار متعدد / صح وخطأ / كتابة)
     * =========================================================================
     */
    function calculateAwardPoints(fishesGroup) {
      const rawSum = fishesGroup.reduce((acc, f) => acc + f.points, 0);
      const rodBonus = (gameState.upgrades.rodLevel - 1) * 5 * fishesGroup.length;
      const boatMultiplier = 1 + (gameState.upgrades.boatLevel - 1) * 0.2;
      return Math.round((rawSum + rodBonus) * boatMultiplier);
    }

    function finalizeQuestionResult(isCorrect, qData, fishesGroup, totalAwardPoints) {
      const feedbackBox = document.getElementById("feedbackBox");
      feedbackBox.classList.remove("hidden");

      const modelAnswerNote = qData.type === "write"
        ? `<div class="mt-1.5 inline-block bg-slate-950/60 border border-yellow-400/40 px-2.5 py-1 rounded-lg text-yellow-300 text-xs">الإجابة النموذجية: <strong>${qData.modelAnswer}</strong></div><br>`
        : "";

      if (isCorrect) {
        sound.correct();
        gameState.coins += totalAwardPoints;

        fishesGroup.forEach(f => {
          f.caught = true;
          markFishCaughtInStorage(gameState.currentLevel, f.index);
          f.el.style.transition = "all 0.35s ease";
          f.el.style.opacity = "0";
          f.el.style.transform = "scale(0.2) translateY(-60px)";
          setTimeout(() => f.el.remove(), 360);
        });

        saveProfile();

        feedbackBox.className = "mt-4 p-3.5 rounded-2xl border border-emerald-400 bg-emerald-950/85 text-emerald-200 text-sm font-bold leading-relaxed";
        feedbackBox.innerHTML = `✅ <strong>إجابة صحيحة! أحسنت يا ${gameState.playerName}! (+${totalAwardPoints} نقطة)</strong><br>${modelAnswerNote}<span class="text-xs text-emerald-100 font-normal">${qData.explanation}</span>`;
      } else {
        sound.wrong();
        fishesGroup.forEach(f => {
          f.el.classList.remove("scale-125");
          f.el.style.top = f.yPct + "%";
        });

        feedbackBox.className = "mt-4 p-3.5 rounded-2xl border border-rose-400 bg-rose-950/85 text-rose-200 text-sm font-bold leading-relaxed";
        feedbackBox.innerHTML = `❌ <strong>إجابة غير مطابقة! عادت السمكة إلى البحر.</strong><br>${modelAnswerNote}<span class="text-xs text-rose-100 font-normal">التوضيح النحوي: ${qData.explanation}</span>`;
      }

      updateHUD();
      const closeBtn = document.getElementById("closeQuestionBtn");
      closeBtn.classList.remove("hidden");
    }

    function renderShop() {
      const container = document.getElementById("shopItemsContainer");
      container.innerHTML = "";
      document.getElementById("shopCoinsDisplay").textContent = gameState.coins;

      SHOP_ITEMS.forEach(item => {
        const curLvl = item.getLevel();
        const isMaxed = curLvl >= item.maxLevel;
        const cost = item.getCost();
        const canAfford = gameState.coins >= cost;

        const card = document.createElement("div");
        card.className = "bg-slate-800/90 border border-slate-700 rounded-2xl p-4 flex flex-col sm:flex-row sm:items-center justify-between gap-3";

        card.innerHTML = `
          <div class="flex-1">
            <div class="flex items-center gap-2 mb-1">
              <h3 class="font-black text-base text-yellow-300">${item.title}</h3>
              <span class="text-[11px] bg-purple-900 border border-purple-500 px-2 py-0.5 rounded-full text-purple-200 font-bold">
                المستوى ${curLvl}/${item.maxLevel}
              </span>
            </div>
            <p class="text-xs text-slate-300 leading-relaxed">${item.desc}</p>
          </div>
          <div class="shrink-0">
            ${
              isMaxed
                ? `<span class="inline-block w-full sm:w-auto text-center px-4 py-2 rounded-xl bg-emerald-900/60 border border-emerald-500 text-emerald-300 font-black text-xs">مُطوّر بالكامل ✔</span>`
                : `<button type="button" data-shop-id="${item.id}" class="w-full sm:w-auto px-4 py-2.5 rounded-xl font-black text-xs sm:text-sm transition cursor-pointer ${
                    canAfford
                      ? "game-btn"
                      : "bg-slate-700 text-slate-400 cursor-not-allowed"
                  }">
                    شراء (${cost} 🪙)
                  </button>`
            }
          </div>
        `;

        container.appendChild(card);
      });

      container.querySelectorAll("button[data-shop-id]").forEach(btn => {
        btn.addEventListener("click", () => {
          const id = btn.getAttribute("data-shop-id");
          const item = SHOP_ITEMS.find(i => i.id === id);
          if (!item) return;

          const cost = item.getCost();
          if (item.getLevel() >= item.maxLevel) return;

          if (gameState.coins < cost) {
            sound.wrong();
            showToast("نقاطك غير كافية! اصطد المزيد من الأسماك أولاً.");
            return;
          }

          gameState.coins -= cost;
          item.apply();
          saveProfile();
          sound.buy();
          showToast(`تم شراء (${item.title}) وحفظه في حسابك! 🎉`);
          updateHUD();
          renderShop();
        });
      });
    }

    window.addEventListener("DOMContentLoaded", () => {
      const startScreen = document.getElementById("startScreen");
      const startBtn = document.getElementById("startGameBtn");
      const nameInput = document.getElementById("playerNameInput");
      const cancelSwitchBtn = document.getElementById("cancelSwitchBtn");

      // التحقق التلقائي من الحساب المحفوظ عند فتح الصفحة
      const hasSavedProfile = loadProfile();
      if (hasSavedProfile) {
        enterGameWorld(true);
      } else {
        startScreen.classList.remove("hidden");
        startScreen.classList.add("flex");
      }

      // إنشاء حساب جديد أو تحديث الاسم
      function createOrSwitchAccount() {
        const entered = nameInput.value.trim();
        if (!entered) {
          showToast("من فضلك اكتب اسمك أولاً ليُحفظ على القارب!");
          nameInput.focus();
          return;
        }

        // إذا كان يبدل إلى اسم جديد مختلف عن الاسم الحالي، نبدأ له حساباً جديداً نظيفاً
        if (gameState.playerName && gameState.playerName !== entered) {
          gameState.coins = 0;
          gameState.currentLevel = 1;
          gameState.unlockedLevel = 1;
          gameState.levelProgress = {};
          gameState.upgrades = {
            rodLevel: 1,
            boatLevel: 1,
            doubleCatch: 0,
            lineDepthLevel: 1
          };
        }

        gameState.playerName = entered;
        saveProfile();
        sound.init();
        enterGameWorld(false);
        showToast(`تم تسجيل وحفظ حساب (${gameState.playerName}) بنجاح! 🔒`);
      }

      startBtn.addEventListener("click", createOrSwitchAccount);
      nameInput.addEventListener("keydown", (e) => {
        if (e.key === "Enter") createOrSwitchAccount();
      });

      // زر تبديل اللاعب في الشريط العلوي
      document.getElementById("switchPlayerBtn").addEventListener("click", (e) => {
        e.stopPropagation();
        nameInput.value = "";
        cancelSwitchBtn.classList.remove("hidden");
        startScreen.classList.remove("hidden");
        startScreen.classList.add("flex");
        setTimeout(() => nameInput.focus(), 80);
      });

      cancelSwitchBtn.addEventListener("click", () => {
        startScreen.classList.add("hidden");
        startScreen.classList.remove("flex");
      });

      // إطلاق الصنارة عند الضغط في منطقة اللعب
      document.getElementById("playArea").addEventListener("click", () => {
        launchHook();
      });

      // زر الصوت
      const soundBtn = document.getElementById("soundToggleBtn");
      soundBtn.addEventListener("click", (e) => {
        e.stopPropagation();
        sound.enabled = !sound.enabled;
        soundBtn.textContent = sound.enabled ? "🔊" : "🔇";
        showToast(sound.enabled ? "تم تفعيل الصوت" : "تم كتم الصوت");
      });

      // قائمة المستويات
      const levelBtn = document.getElementById("levelDropdownBtn");
      const levelMenu = document.getElementById("levelMenu");
      levelBtn.addEventListener("click", (e) => {
        e.stopPropagation();
        levelMenu.classList.toggle("hidden");
      });

      document.addEventListener("click", (e) => {
        if (!levelBtn.contains(e.target) && !levelMenu.contains(e.target)) {
          levelMenu.classList.add("hidden");
        }
      });

      // نافذة المتجر
      document.getElementById("openShopBtn").addEventListener("click", (e) => {
        e.stopPropagation();
        renderShop();
        const shopModal = document.getElementById("shopModal");
        shopModal.classList.remove("hidden");
        shopModal.classList.add("flex");
      });

      document.getElementById("closeShopBtn").addEventListener("click", () => {
        const shopModal = document.getElementById("shopModal");
        shopModal.classList.add("hidden");
        shopModal.classList.remove("flex");
      });
